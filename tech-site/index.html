// 三行代码，接入 NEURA 决策引擎
import { Neura } from '@neura/sdk';

const engine = new Neura({
  apiKey: process.env.NEURA_KEY,
  region: 'cn-east-1',
  autoScale: true,
});

// 实时推理，8ms 内返回决策结果
const result = await engine.decide({
  context: userSession,
  objective: 'maximize_conversion',
});

console.log(result.confidence); // → 0.997
